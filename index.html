⚡ Code Cloudflare Worker (Gratuit)

Steam + IGDB

Créez un Worker gratuit sur workers.cloudflare.com, ajoutez vos variables d'environnement TWITCH_CLIENT_ID et TWITCH_TOKEN (API IGDB), et collez ce code : 

export default {
  async fetch(req, env) {
    const cors = { "Access-Control-Allow-Origin": "*", "Access-Control-Allow-Headers": "Content-Type", "Content-Type": "application/json" };
    if (req.method === "OPTIONS") return new Response(null, { headers: cors });
    const url = new URL(req.url);

    // 1. Récupérer la wishlist Steam publique
    if (url.searchParams.has("steamid")) {
      const id = url.searchParams.get("steamid");
      const r = await fetch(`https://api.steampowered.com/IWishlistService/GetWishlist/v1/?steamid=${id}`);
      return new Response(await r.text(), { headers: cors });
    }

    // 2. Vérifier un jeu sur IGDB (PS5 = 167, Switch = 130, Switch 2 = 508)
    const { title } = await req.json();
    const igdb = await fetch("https://api.igdb.com/v4/games", {
      method: "POST",
      headers: { "Client-ID": env.TWITCH_CLIENT_ID, "Authorization": `Bearer ${env.TWITCH_TOKEN}` },
      body: `search "${title.replace(/"/g, '')}"; fields name,platforms; limit 1;`
    });
    const [game] = await igdb.json();
    const p = game?.platforms || [];
    return new Response(JSON.stringify({
      ps5: p.includes(167) ? "yes" : "no",
      switch1: p.includes(130) ? "yes" : "no",
      switch2: p.includes(508) ? "yes" : "no"
    }), { headers: cors });
  }
};
